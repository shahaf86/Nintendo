services:
  db:
    image: postgres:16-alpine
    container_name: lore-db
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-lore}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-lore_dev}
      POSTGRES_DB: ${POSTGRES_DB:-lore}
    ports:
      - "${POSTGRES_PORT:-5432}:5432"
    volumes:
      - lore_pgdata:/var/lib/postgresql/data
      # Everything here runs, in filename order, on FIRST boot of the volume only.
      - ./db/init:/docker-entrypoint-initdb.d:ro
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-lore} -d ${POSTGRES_DB:-lore}"]
      interval: 5s
      timeout: 5s
      retries: 10

  # Optional GUI. Start with:  docker compose --profile tools up -d
  pgadmin:
    image: dpage/pgadmin4:latest
    container_name: lore-pgadmin
    profiles: ["tools"]
    depends_on:
      db:
        condition: service_healthy
    environment:
      PGADMIN_DEFAULT_EMAIL: dev@local.test
      PGADMIN_DEFAULT_PASSWORD: admin
      PGADMIN_CONFIG_SERVER_MODE: "False"
    ports:
      - "5050:80"

volumes:
  lore_pgdata:
